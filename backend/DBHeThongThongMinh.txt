Table users {
  id int [pk, increment]
  username varchar(50) [unique, not null]
  password varchar(255) [not null]
  email varchar(100) [unique, not null]
  address text
  created_at timestamp [default: `now()`]
}

Table carts {
  id int [pk, increment]
  user_id int [unique, not null]
  created_at timestamp [default: `now()`]
}

Table cart_items {
  id int [pk, increment]
  cart_id int [not null]
  product_id int [not null]
  quantity int [not null, default: 1]
  created_at timestamp [default: `now()`]
}

Table shipping_methods {
  id int [pk, increment]
  name varchar(100) [not null]
  fee decimal(15, 2) [not null, default: 0]
  estimated_days varchar(50)
  is_active boolean [default: true]
  created_at timestamp [default: `now()`]
}

Table orders {
  id int [pk, increment]
  user_id int [not null]
  shipping_method_id int [not null]
  total_amount decimal(15, 2) [not null, default: 0]
  status varchar(50) [not null, default: 'pending']
  shipping_address text [not null]
  shipping_phone varchar(20) [not null]
  created_at timestamp [default: `now()`]
}

Table order_items {
  id int [pk, increment]
  order_id int [not null]
  product_id int [not null]
  quantity int [not null, default: 1]
  price decimal(15, 2) [not null]
  created_at timestamp [default: `now()`]
}

Table categories {
  id int [pk, increment]
  name varchar(100) [not null]
}

Table products {
  id int [pk, increment]
  category_id int [not null]
  name varchar(255) [not null]
  color varchar(50)
  stock_quantity int [not null, default: 0]
  image_url text [not null]
  price decimal(15, 2) [not null, default: 0]
  rating decimal(3, 2) [default: 0]
  created_at timestamp [default: `now()`]
}

Table mice {
  id int [pk, increment]
  product_id int [unique, not null]
  brand varchar(100) [not null]
  connectivity varchar(50) [not null] // 'wired', 'wireless', 'tri_mode'
  hand_orientation varchar(20) [not null] // 'symmetrical', 'ergonomic'
  weight_g decimal(5,1) [not null]
  polling_rate_hz int [not null, default: 1000]
  switch_type varchar(100)
  sensor varchar(100)
  created_at timestamp [default: `now()`]
}

Table keyboards {
  id int [pk, increment]
  product_id int [unique, not null]
  brand varchar(100) [not null]
  connectivity varchar(50) [not null] // 'wired', 'wireless', 'tri_mode'
  layout varchar(50) [not null] // '60%', '65%', '75%', 'TKL', 'Fullsize'
  has_rapid_trigger boolean [default: false]
  has_hotswap boolean [default: true]
  support_macos boolean [default: false]
  polling_rate_hz int [not null, default: 1000]
  switch varchar(100)
  created_at timestamp [default: `now()`]
}

Table headphones {
  id int [pk, increment]
  product_id int [unique, not null]
  brand varchar(100) [not null]
  connectivity varchar(50) [not null] // 'wired_3.5mm', 'wireless_2.4g', 'bluetooth'
  headphone_type varchar(50) [not null] // 'in_ear', 'over_ear'
  has_mic boolean [default: true]
  has_anc boolean [default: false]
  created_at timestamp [default: `now()`]
}

Table setups {
  id int [pk, increment]
  product_id int [unique, not null]
  brand varchar(100) [not null]
  setup_category varchar(50) [not null] // 'chair', 'desk', 'arm'
  max_load_kg int
  created_at timestamp [default: `now()`]
}

Table mousepads {
  id int [pk, increment]
  product_id int [unique, not null]
  brand varchar(100) [not null]          // Thương hiệu[cite: 1]
  pad_type varchar(50)                   // Loại sản phẩm[cite: 1]
  mouse_control varchar(50)              // Kiểm soát chuột[cite: 1]
  speed varchar(50)                      // Speed[cite: 1]
  surface varchar(100)                   // Bề mặt[cite: 1]
  base varchar(100)                      // Đế[cite: 1]
  size varchar(50)                       // Kích thước[cite: 1]
  created_at timestamp [default: `now()`]
}
